The script to run once Bash or Python is available (save outside the data folder, run with `python3 -I`):

```python
import csv, re, sys
from collections import Counter
PREF = ["CenterPoint","Northern Indiana Public Service","NIPSCO","Citizens Gas","Citizens Energy",
        "Vectren","Indiana American Water","Ohio Valley Gas","American Electric Power","Duke Energy",
        "Community Natural Gas","Midwest Natural Gas","Frontier Communications"]
SUF = {"inc","llc","co","corp","corporation","company"}
def is_util(r):
    s = r.lstrip().casefold()
    return any(s.startswith(p.casefold()) for p in PREF)
def norm(name):
    s = "".join(c for c in name.lower() if c.isalpha() or c.isdigit() or c == " ")
    parts = re.sub(" +", " ", s).strip().split(" ")
    if parts and parts[-1] in SUF: parts = parts[:-1]
    return " ".join(parts)
rows = [r for r in csv.DictReader(open(sys.argv[1], newline="", encoding="utf-8"))
        if "2021-01-01" <= r["order_date"] <= "2025-12-31"]
nu = [r for r in rows if not is_util(r["respondent"])]
c = Counter(norm(r["respondent"]) for r in nu)
big = {k: v for k, v in c.items() if v >= 5}
print("non-utility cases:", len(nu), "| distinct:", len(c), "| >=5:", len(big))
print("cases covered:", sum(big.values()), "| share %:", round(100*sum(big.values())/len(nu), 1))
```
